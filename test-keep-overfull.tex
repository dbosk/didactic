% Test that a page made overfull by what can't part from the start of a listing
% warns, whatever kind of listing it is, and names the place (didactic).
%
% Build: see the Makefile; test-check.py checks that each case below warns, with
% its input line (the line in this file, not in the listing's file or minted's) and
% the page that is overfull, from the label.  The log has an Overfull \vbox for each
% case but P: marginfix re-wraps the page in \@outputbox, whose depth is real, and
% \@makecol's \vskip-\dimen@ takes it all off, so the pair's great depth gives no
% message although the page runs off the paper.
%   V  A listing kept whole, by \VerbatimInput.
%   E  The same by \lstexample*, whose lines come from minted's file.
%   L  A long listing, 30 lines: only its first lines can't part from the lead-in.
%   T  A listing made long by the height of its lines.
%   P  Code side by side, one box.
%   F  The same as V, with \newpage for \clearpage before it, so that the float page is
%      shipped out right after the full page, which the warning must name.
%   C  Control: the same as V, in room: no warning.
\documentclass[a4paper,oneside]{memoir}
\usepackage[british]{babel}
\usepackage{amsmath,amssymb,amsthm}
% With the limit of keepmaxgap the tie is released before the page is overfull, so
% this test turns the limit off, to reach the case it is about.
\PassOptionsToPackage{keeplines=20,keepleadin=true,keepmaxgap=0}{didactic}
\usepackage{didactic}
\usepackage[noamsthm,notheorems]{beamerarticle}
\usepackage{refcount}
\makeatletter
\newwrite\@ofull@out
\newcount\@ofull@i
\newcommand\testfile[3]{%
  \immediate\openout\@ofull@out=\jobname-#1.txt
  \@ofull@i\z@
  \@whilenum\@ofull@i<#2\do{\advance\@ofull@i\@ne
    \immediate\write\@ofull@out{#3}}%
  \immediate\closeout\@ofull@out}
\newcommand\case[1]{\typeout{overfull case #1 line \the\inputlineno}}
\newcommand\results[1]{\typeout{overfull result #1 page=\getpagerefnumber{#1}}}
\makeatother
\begin{document}

\testfile{v}{10}{verbatim}%
\case{v}\VerbatimInput{\jobname-v.txt}
