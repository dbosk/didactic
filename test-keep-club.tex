% The club and widow penalties stay as the class has them (didactic), after a heading
% and a list too, the maintainer's decision: 1000 under memoir, and 150 under article,
% which test-keep-club-article reads this with.
\ifdefined\testclass\else\def\testclass{memoir}\def\testpen{1000}\fi
\documentclass[a4paper]{\testclass}
\usepackage{amsmath,amssymb,amsthm}
\usepackage{didactic}
\makeatletter
\newcommand\checkpen[1]{%
  \ifnum\clubpenalty=\testpen\relax \ifnum\widowpenalty=\testpen\relax
    \typeout{didactic club test: PASS #1}%
  \else\typeout{didactic club test: FAIL #1, widow \the\widowpenalty}\fi
  \else\typeout{didactic club test: FAIL #1, club \the\clubpenalty}\fi}
\makeatother
\begin{document}
Before any heading.\checkpen{before a heading}\par

\section{A heading}
First paragraph.\par
Second paragraph.\checkpen{after a heading}\par
\begin{itemize}
\item An item.
\end{itemize}
After a list.\checkpen{after a list}\par
\typeout{didactic club test: all checks ran}
\end{document}
