% Test of keeping listings together (didactic): in the notes (memoir +
% beamerarticle), a listing of at most keeplines lines doesn't break across
% pages, a longer one leaves at least keepends lines on each side of a
% break, and the paragraph that leads into a listing stays with it.  This document
% pins keepends=3; the default of 2 is in test-keep-heading and test-keep-sweep.
% Each case starts on a new page, after a rule that leaves room for a few
% lines only, so that without didactic the listing would break there.
%
% Build: see the Makefile (LaTeX, PythonTeX, LaTeX until the page references
% settle).  The
% document checks itself: the last page lists every check with PASS or
% FAIL, and the log has a line "didactic keep test: PASS/FAIL <check>",
% the result first as TeX wraps long lines in the log,
% for each, then "didactic keep test: all checks ran"; the Makefile fails
% on any FAIL or SKIP and when that last line is missing.
%
% Expected in the resulting PDF, besides PASS everywhere:
%   A  a short listing (10 lines) with room for 4: the lead-in and the
%      whole listing on the next page, the first page empty below the rule.
%   B  a long listing (30 lines) with room for the lead-in and 2 lines:
%      the lead-in and the first three lines on the next page.
%   C  a long listing with room for 10 lines: it breaks there.
%   D  a long listing with room for all but one line: the last three
%      lines on the next page, not one.
%   I  as D, but line 28 is empty: lines 27 to 30 on the next page, so
%      that three lines that aren't empty are there and the page doesn't
%      start with the empty line.
%   W  as I, but line 29 holds only spaces, in a minted listing whose lexer
%      keeps them (bash): lines 27 to 30 on the next page.  The Makefile
%      checks that from the PDF, since it has no labels.
%   E  a short listing read by \inputminted (5 lines), with room for 3
%      lines: on the next page with its lead-in.
%   F  a code chunk woven for minted (as noweb.sty's [minted] option and
%      the tominted weave give), with room for 2 lines: the lead-in, the
%      header line "<hello.py>=" and the code on the next page.
%      Without noweb.sty it is a SKIP, which fails the build.
%   J  a pair of didacticpair (as \codebycode and \runbyrun make), 8 lines
%      tall, with room for 2 lines: on the next page with its lead-in.
%   S  a short listing over the full width by \lstexample*, with room for
%      2 lines: on the next page with its lead-in.
%   Y  the same by didactic_mint(fullwidth=True), through PythonTeX.
%   S2 as S with room for 5.5 lines and a lead-in of three, where moving them
%      would leave more than keepmaxgap: the listing starts the next page, alone.
%   N  a listing after a paragraph with an \ltnote in it, with room for 2
%      lines: on the next page with its lead-in.  M the same with \marginpar,
%      P with a float before the lead-in.  (Between the lead-in and the listing,
%      on their own after a blank line, they give a place to break, which the
%      docs state as a limit.  Taking that place away isn't right either:
%      test-keep-chain shows paragraphs that can't part.)
%   O  a listing with formatcom=\color{blue}: on the next page with its
%      lead-in.
%   K  a long listing with line 13 empty and room for 12 lines: lines 12 up
%      on the next page, so that the page doesn't start with the empty line.
%   L  a long listing with line 29 empty, so the last lines are 27, 28 and
%      30: line 27 on the same page as line 30, empty lines don't count.
%   G  a short listing in an item of a list, room for 3 lines: on the next
%      page with the item's first line.
%   Q  a listing that starts an item: the item's label "1." beside the
%      first line, which the Makefile checks from the PDF.
%   H  the depth after a listing is that of its last line, not of what came
%      before it, so that the text after it is at the right height.  (The
%      lead-in has a depth of 7pt, the listing's last line almost none.)
%   T  a short listing (10 lines) with room for 8 of its lines: moving it
%      would leave more than keepmaxgap (5) lines empty, so it breaks, with
%      at least three lines on each side.
%   U  a section heading followed by four lines of text, with room for it
%      and one line, within keepmaxgap: it moves to the next page with three
%      lines (keepafterheading).
%   X  a place to break of noweb between two paragraphs (\nwdocspar), with room
%      for a few lines of the second: with keepmaxgap it isn't taken, the text
%      goes on on this page.
%   Z  a listing of 15 lines, over the default of keeplines, with room for 4:
%      it breaks, rather than moving as a whole.
%   V  the heading of a block (example), with room for one of its lines:
%      the block moves to the next page.
%   R  a section heading, a sentence and a longtable whose caption and
%      first row don't fit, with room for little: the heading moves to the page
%      of the table, as the estimate before it sees.  (When only the table's
%      place shows it, in the next run: test-keep-heading, AB to AD.)

% \roomfor{n}: start a new page and fill it so that n lines of text fit
% under the rule.
\newcommand\roomfor[1]{%
  \clearpage
  \noindent\rule{\linewidth}{0.4pt}\par
  \vspace*{\dimexpr\textheight-\topskip-#1\baselineskip-\baselineskip\relax}%
  \par}
% \testfile{prefix}{n}{before}{after}: write the file \jobname-prefix.txt
% of n lines "line <i>", each followed by
% <before>\label{prefix L<i>}<after>, except an empty line set by
% \testblank.
\newcount\@keeptest@i
\newcommand\testfile[4]{%
  \immediate\openout\@keeptest@out=\jobname-#1.txt
  \@keeptest@i\z@
  \loop
    \advance\@keeptest@i\@ne
    \ifnum\@keeptest@i=\@keeptest@blank
      \immediate\write\@keeptest@out{\@keeptest@blankline}%
    \else
      \immediate\write\@keeptest@out{%
        line \the\@keeptest@i#3\string\label{#1L\the\@keeptest@i}#4}%
    \fi
  \ifnum\@keeptest@i<#2\repeat
  \immediate\closeout\@keeptest@out
  \global\@keeptest@blank\z@
  \global\let\@keeptest@blankline\@empty}
% \testplainfile{prefix}{n}: such a file of lines "<prefix>line <i>" without
% labels, since escapeinside fails in some lexers.  The Makefile checks these
% from the PDF.
\newcommand\testplainfile[2]{%
  \immediate\openout\@keeptest@out=\jobname-#1.txt
  \@keeptest@i\z@
  \@whilenum\@keeptest@i<#2\do{%
    \advance\@keeptest@i\@ne
    \ifnum\@keeptest@i=\@keeptest@blank
      \immediate\write\@keeptest@out{\@keeptest@blankline}%
    \else
      \immediate\write\@keeptest@out{#1line \the\@keeptest@i}%
    \fi}%
  \immediate\closeout\@keeptest@out
  \global\@keeptest@blank\z@
  \global\let\@keeptest@blankline\@empty}
% \testblank{i}: line i of the next file is empty; \testspaces{i}: it holds
% only spaces.
\newcount\@keeptest@blank
\let\@keeptest@blankline\@empty
\newcommand\testblank[1]{\global\@keeptest@blank=#1\relax}
\newcommand\testspaces[1]{%
  \testblank{#1}%
  \gdef\@keeptest@blankline{\space\space\space\space}}
% \testlisting{prefix}{n}: such a file as a listing of fancyvrb.
\newcommand\testlisting[2]{%
  \testfile{#1}{#2}{}{}%
  \VerbatimInput[commandchars=\\\{\}]{\jobname-#1.txt}}
\newwrite\@keeptest@out
% \testcheck{name}{label1}{=|<}{label2}: are the labels on the same page (=)
% or is the first on an earlier page (<)?
% A label that doesn't exist has page 0: that is a FAIL, not a match.
\newcommand\testcheck[4]{%
  \def\@keeptest@result{FAIL}%
  \ifnum\getpagerefnumber{#2}>\z@
    \ifnum\getpagerefnumber{#4}>\z@
      \ifnum\getpagerefnumber{#2}#3\getpagerefnumber{#4}\relax
        \def\@keeptest@result{PASS}%
      \fi
    \fi
  \fi
  \typeout{didactic keep test: \@keeptest@result\space #1}%
  \item #1: \@keeptest@result\ (#2 on page \getpagerefnumber{#2}, #4 on
    page \getpagerefnumber{#4})}
\makeatother

\section*{Results}
\begin{itemize}
\item[] A short listing doesn't break and stays with its lead-in:
\testcheck{A first line with lead-in}{a-lead}{=}{aL1}
\testcheck{A first line on the next page}{a-lead}{=}{aL10}
\item[] A long listing keeps three lines at each end:
\testcheck{B lead-in with line 3}{b-lead}{=}{bL3}
\testcheck{C breaks}{cL1}{<}{cL30}
\testcheck{C lead-in with line 3}{c-lead}{=}{cL3}
\testcheck{C line 28 with the last}{cL28}{=}{cL30}
\testcheck{D breaks}{dL1}{<}{dL30}
\testcheck{D line 28 with the last}{dL28}{=}{dL30}
\testcheck{D line 27 with the first}{dL1}{=}{dL27}
\testcheck{I line 27 with the last}{iL27}{=}{iL30}
\testcheck{I breaks}{iL1}{<}{iL30}
\item[] minted, noweb and lists:
\testcheck{E lead-in with the listing}{e-lead}{=}{eL5}
\makeatletter
\ifdefined\nwbegincode
\makeatother
\testcheck{F lead-in with the header and code}{f-lead}{=}{fL1}
\testcheck{F the whole chunk}{f-lead}{=}{fL5}
\else
\makeatother
\typeout{didactic keep test: SKIP F no noweb.sty}%
\item F: SKIP, no noweb.sty
\fi

\testcheck{G lead-in with the listing}{g-lead}{=}{gL1}
\testcheck{K breaks}{kL1}{<}{kL30}
\testcheck{K no page starts with the empty line}{kL1}{<}{kL12}
\testcheck{K line 12 with line 14}{kL12}{=}{kL14}
\testcheck{L breaks}{lL1}{<}{lL30}
