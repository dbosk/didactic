% Test that a listing that starts a column apart from its lead-in warns (didactic):
% no page is shipped out between the columns, and only \outputpenalty knows the break.
%
% Build: see the Makefile; test-check.py checks the warnings.
%   M  A lead-in at the foot of the first column, a note, and a listing: the listing
%      starts the second column, and warns.
%   C  Control: a lead-in and a listing with nothing between: they go along.
%   O  A column overfull with what can't part: no page is shipped out at the end of the
%      column, so the warning names the page of the listing, not the last page shipped.
%      The \nobreak leaves no place to break before the lead-in, which would fill the
%      first column with nothing and ship out the page.
%   R  The same with code side by side.
\documentclass[a4paper,twocolumn]{memoir}
\usepackage[british]{babel}
\usepackage{amsmath,amssymb,amsthm}
\usepackage{didactic}
\usepackage[noamsthm,notheorems]{beamerarticle}
\newcommand\case[1]{\typeout{twocol case #1 line \the\inputlineno}}
\begin{document}
\noindent\rule{1pt}{\dimexpr\textheight-\topskip-3\baselineskip\relax}\par
LEADM in.

\marginpar{note M}
\case{m}\begin{Verbatim}
LISTM
line 2
line 3
\end{Verbatim}
\clearpage
\noindent\rule{1pt}{\dimexpr\textheight-\topskip-3\baselineskip\relax}\par
LEADC in.

\case{c}\begin{Verbatim}
LISTC
line 2
line 3
\end{Verbatim}
\clearpage
\nobreak
Tall lead-in O in. \rule{1pt}{0.9\textheight}
\case{o}\begin{Verbatim}
line 1
line 2
line 3
line 4
line 5
line 6
line 7
line 8
line 9
line 10
\end{Verbatim}
\clearpage
\nobreak
Tall lead-in R in. \rule{1pt}{0.9\textheight}
\case{r}\begin{didacticpair}{2}
  \rule{1pt}{8\baselineskip} & right
\end{didacticpair}
\end{document}
