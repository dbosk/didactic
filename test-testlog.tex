% and runs testlog.py selftest on each log, which also checks that each phrase
% is cut by a break there, so that the test still tests the joiner.  Keep the
% messages as they are: their columns are what is tested.

\begin{document}

  (found from the last run): fixture}

  isn't plain: it has space of its own at its end}

  kurs/del-ett/text/kapitel-översikt.tex:12, in the margin of page 3}

  (CCS) is at the foot of page 4}
