% Test of frame titles (didactic) in the notes: several titles in a frame.  The
% title waits for the next paragraph, and by then the frame may have another
% title.
%
% Expected in the resulting PDF:
%   1  OVERA, then OVERB with TEXTOVER on its line: each title once, with the
%      overlay specification.
%   2  The same for ONLYA and ONLYB, in \only.
\documentclass[a4paper]{memoir}
\usepackage{amsmath,amssymb,amsthm}
\usepackage{didactic}
\usepackage[noamsthm,notheorems]{beamerarticle}
\begin{document}
\begin{frame}
  \frametitle<1>{OVERA}
  \frametitle<2>{OVERB}
  TEXTOVER has two titles.
\end{frame}
\begin{frame}
  \only<1>{\frametitle{ONLYA}}
  \only<2>{\frametitle{ONLYB}}
  TEXTONLY has two titles.
\end{frame}
\end{document}
